/* zxregex.c — small POSIX ERE bridge for Zexi's opt-in regex module.
 *
 * This deliberately uses the system POSIX regex implementation instead of
 * shipping a large PCRE-like runtime. It is linked only for `use regex;` /
 * `import text.regex;` programs.
 */
#include <stdint.h>
#include <regex.h>
#include <string.h>

/* Keep this generated-C-compatible name and layout. The bridge source is a
 * separate translation unit, so it does not need the generated header. */
typedef struct zx_RegexSpan {
    int64_t start;
    int64_t end;
} zx_RegexSpan;

int zxr_is_valid(const char *pattern) {
    if (!pattern) return 0;
    regex_t re;
    int ok = regcomp(&re, pattern, REG_EXTENDED) == 0;
    if (ok) regfree(&re);
    return ok;
}

/* zxr_search_span returns 1 and fills the first matched byte range, or 0 for
 * an invalid pattern/no match. The generated Zexi ABI passes a compatible
 * two-i64 RegexSpan structure by pointer. */
int zxr_search_span(const char *pattern, const char *text, zx_RegexSpan *span) {
    if (!span) return 0;
    span->start = -1;
    span->end = -1;
    if (!pattern || !text) return 0;

    regex_t re;
    if (regcomp(&re, pattern, REG_EXTENDED) != 0) return 0;
    regmatch_t match;
    int found = regexec(&re, text, 1, &match, 0) == 0;
    if (found && match.rm_so >= 0 && match.rm_eo >= match.rm_so) {
        span->start = (int64_t)match.rm_so;
        span->end = (int64_t)match.rm_eo;
    } else {
        found = 0;
    }
    regfree(&re);
    return found;
}

int zxr_full_match(const char *pattern, const char *text) {
    if (!pattern || !text) return 0;
    regex_t re;
    if (regcomp(&re, pattern, REG_EXTENDED) != 0) return 0;
    regmatch_t match;
    int found = regexec(&re, text, 1, &match, 0) == 0;
    int full = found && match.rm_so == 0 && match.rm_eo == (regoff_t)strlen(text);
    regfree(&re);
    return full ? 1 : 0;
}
